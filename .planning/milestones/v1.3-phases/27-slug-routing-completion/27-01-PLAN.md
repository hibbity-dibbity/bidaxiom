    3. Add the update mutation — **CRITICAL: use `onSuccess(updated)` to capture the returned record and navigate to the new slug**:
       ```tsx
       const updateMutation = trpc.customer.update.useMutation({
         onSuccess: (updated) => {
           utils.customer.getBySlug.invalidate({ slug: customerSlug });
           utils.customer.list.invalidate();
           toast.success("Customer updated successfully");
           setEditDialogOpen(false);
           // If name changed, slug regenerated — navigate to new slug URL
           if (updated.slug && updated.slug !== customerSlug) {
             router.replace(`/customers/${updated.slug}`);
           }
         },
         onError: (error) => {
           toast.error(error.message || "Failed to update customer");
         },
       });
       ```

           phone: customer.phone || "",
           email: customer.email || "",
           address: {
             street1: customer.street1 || "",
             street2: customer.street2 || "",
             city: customer.city || "",
             state: customer.state || "",
             zipCode: customer.zip || "",

       const handleSubmitEdit = (data: CustomerFormData) => {
         if (!customer) return;

           contactName: data.contactName,
           phone: data.phone,
           email: data.email,
           address: data.address,
           street1: data.address?.street1,

           city: data.address?.city,
           state: data.address?.state,
           zip: data.address?.zipCode,

       };
