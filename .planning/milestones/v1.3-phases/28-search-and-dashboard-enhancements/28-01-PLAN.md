```typescript
const estimateResults = await ctx.tenant.db
  .select({
    id: estimates.id,
    name: estimates.name,
    status: estimates.status,
    slug: estimates.slug,
    projectSlug: projects.slug,
    projectName: projects.name,
  })
  .from(estimates)
  .innerJoin(projects, eq(estimates.projectId, projects.id))
  .where(

      isNotNull(estimates.slug)
    )
  )
  .limit(limit);
```
