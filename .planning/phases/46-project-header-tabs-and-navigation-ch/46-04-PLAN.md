```ts
function formatAddress(street1: string | null | undefined, city: string | null | undefined, state: string | null | undefined, zip: string | null | undefined): string | null {
  const parts = [street1, [city, state].filter(Boolean).join(", "), zip].filter(Boolean);
  return parts.length > 0 ? parts.join(" ") : null;
}

function formatDateRange(startDate: string | null | undefined, endDate: string | null | undefined): string | null {
  if (!startDate) return null;
  const start = new Date(startDate).toLocaleDateString("en-US", { month: "short", year: "numeric" });
  if (!endDate) return start;
  const end = new Date(endDate).toLocaleDateString("en-US", { month: "short", year: "numeric" });
  return `${start} – ${end}`;
}
```
