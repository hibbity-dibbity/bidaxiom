In `apps/api/src/server.ts`, register a new Fastify route (scoped plugin, same pattern as the webhook plugin — NOT using `fp()`) for `POST /api/documents/bulk-download`:

```typescript
// In the existing server setup, after other routes are registered:
fastify.post('/api/documents/bulk-download', {
  onRequest: [fastify.authenticate],  // reuse existing auth middleware
}, async (request, reply) => {
  // Body: { documentIds: string[], tenantId: string }
  const { documentIds, tenantId } = request.body as { documentIds: string[]; tenantId: string };

  if (!Array.isArray(documentIds) || documentIds.length === 0) {
    return reply.code(400).send({ error: 'documentIds required' });
  }
  if (documentIds.length > 50) {
    return reply.code(400).send({ error: 'Maximum 50 documents per bulk download' });
  }

  // Get tenant DB from request context (same pattern as tRPC procedures — ctx.tenant)
  const tenant = getTenantContext(tenantId); // use existing tenant resolution
  const { db, schema } = tenant;

  // Fetch document + version records for all requested IDs
  const docs = await db
    .select({
      id: schema.documents.id,
      name: schema.documents.name,
      currentVersionId: schema.documents.currentVersionId,
    })
    .from(schema.documents)
    .where(inArray(schema.documents.id, documentIds));

  // Fetch S3 keys from document_versions
  const versionIds = docs.map(d => d.currentVersionId).filter(Boolean) as string[];
  const versions = await db
    .select({ id: schema.documentVersions.id, storageKey: schema.documentVersions.storageKey })
    .from(schema.documentVersions)
    .where(inArray(schema.documentVersions.id, versionIds));

  const keyMap = new Map(versions.map(v => [v.id, v.storageKey]));

  reply.header('Content-Type', 'application/zip');
  reply.header('Content-Disposition', 'attachment; filename="documents.zip"');

  const archive = archiver('zip', { zlib: { level: 6 } });
  archive.pipe(reply.raw);

  for (const doc of docs) {
    const storageKey = doc.currentVersionId ? keyMap.get(doc.currentVersionId) : null;
    if (!storageKey) continue;
    const s3Response = await s3Client.send(new GetObjectCommand({
      Bucket: process.env.AWS_S3_BUCKET!,
      Key: storageKey,
    }));
    if (s3Response.Body) {
      const { Readable } = await import('stream');
      archive.append(Readable.from(s3Response.Body as any), { name: doc.name });
    }
  }

  await archive.finalize();
});
```
