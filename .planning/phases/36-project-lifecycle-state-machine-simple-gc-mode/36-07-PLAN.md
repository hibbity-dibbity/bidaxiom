**Step 1 — Add helper to packages/shared/src/types/project.ts:**

Read the file first to find the correct location (after existing lifecycle phase constants). Then add:

```typescript
// Phase-feature gate definitions
// feature -> list of lifecyclePhase values where the feature IS available
export const PHASE_FEATURE_GATES: Record<string, LifecyclePhase[]> = {
  change_orders: ["awarded", "active", "substantial_completion", "closed", "warranty", "archived"],
  estimate_editing: ["opportunity", "estimating"],
};

/**
 * Returns true if the given feature is accessible in the given lifecycle phase.
 * Used in both tRPC mutation guards (server) and tab visibility (client).
 *
 * @param feature  - key from PHASE_FEATURE_GATES (e.g. 'change_orders')
 * @param phase    - current lifecyclePhase of the project
 */
export function isFeatureAvailableInPhase(feature: string, phase: LifecyclePhase): boolean {
  const allowed = PHASE_FEATURE_GATES[feature];
  if (!allowed) return true; // unknown feature keys are permissive by default
  return allowed.includes(phase);
}
```

Make sure `LifecyclePhase` type is already exported from this file (it is defined as a union or enum in Plan 01). Import or reference it accordingly. Export both `PHASE_FEATURE_GATES` and `isFeatureAvailableInPhase` from the package's index.ts barrel if a barrel file exists.

```typescript
import { isFeatureAvailableInPhase } from "@construction/shared";

// Inside create mutation, after resolving projectId:
const [project] = await db
  .select({ lifecyclePhase: schema.projects.lifecyclePhase })
  .from(schema.projects)
  .where(eq(schema.projects.id, input.projectId))
  .limit(1);

if (!project) throw new TRPCError({ code: "NOT_FOUND", message: "Project not found" });

if (!isFeatureAvailableInPhase("change_orders", project.lifecyclePhase as LifecyclePhase)) {
  throw new TRPCError({
    code: "BAD_REQUEST",
    message: `Change Orders are not available in the '${project.lifecyclePhase}' phase. Projects must reach the Awarded phase before change orders can be created.`,
  });
}
```
