const prefetchBudget = () => {
  if (!projectId) return;
  utils.budget.getSummary.prefetch({ projectId });
  utils.budget.getAlerts.prefetch({ projectId });
  utils.budget.getForecast.prefetch({ projectId });
};

const prefetchDocuments = () => {
  if (!projectId) return;
  utils.document.list.prefetch({ projectId });
};

const prefetchChangeOrders = () => {
  if (!projectId) return;
  utils.changeOrder.list.prefetch({ projectId });
};

const prefetchCommitments = () => {
  if (!projectId) return;
  utils.commitment.list.prefetch({ projectId });
};

const prefetchSOV = () => {
  if (!projectId) return;
  utils.sov.getItems.prefetch({ projectId });
};

const prefetchPaymentApps = () => {
  if (!projectId) return;
  utils.paymentApp.list.prefetch({ projectId });
};

const prefetchRFQs = () => {
  if (!projectId) return;
  utils.rfq.list.prefetch({ projectId });
};

4. Add onMouseEnter handlers to tab links:
```typescript
const tabs = [
  { id: 'overview', label: 'Overview', href: `/projects/${projectSlug}`, prefetch: () => {} },
  { id: 'documents', label: 'Documents', href: `/projects/${projectSlug}/documents`, prefetch: prefetchDocuments },
  { id: 'budget', label: 'Budget', href: `/projects/${projectSlug}/budget`, prefetch: prefetchBudget },
  { id: 'commitments', label: 'Commitments', href: `/projects/${projectSlug}/commitments`, prefetch: prefetchCommitments },
  { id: 'change-orders', label: 'Change Orders', href: `/projects/${projectSlug}/change-orders`, prefetch: prefetchChangeOrders },
  { id: 'sov', label: 'SOV', href: `/projects/${projectSlug}/sov`, prefetch: prefetchSOV },
  { id: 'payment-apps', label: 'Payment Apps', href: `/projects/${projectSlug}/payment-apps`, prefetch: prefetchPaymentApps },
  { id: 'invoices', label: 'Invoices', href: `/projects/${projectSlug}/invoices`, prefetch: prefetchInvoices },
  { id: 'rfqs', label: 'RFQs', href: `/projects/${projectSlug}/rfqs`, prefetch: prefetchRFQs },
];

return (
  <div className="border-b">
    <nav className="flex gap-6">
      {tabs.map((tab) => (
        <Link
          key={tab.id}
          href={tab.href}
          onMouseEnter={tab.prefetch} // Prefetch on hover
          className={cn(
            "border-b-2 px-1 pb-3 text-sm font-medium transition-colors",
            activeTab === tab.id
              ? "border-primary text-foreground"
              : "border-transparent text-muted-foreground hover:text-foreground"

        >
          {tab.label}
        </Link>
      ))}
    </nav>
  </div>
);
```
