```typescript
const exportCsv = () => {
  const today = new Date().toISOString().split("T")[0];
  const filename = `estimates-${projectSlug}-${today}.csv`;

  const escape = (v: string) => `"${String(v).replace(/"/g, '""')}"`;

  const header = ["Name", "Status", "Total Cost", "Created Date"].map(escape).join(",");
  const csvRows = displayedEstimates.map((e) => [
    e.name,
    e.status.charAt(0).toUpperCase() + e.status.slice(1),
    Number(e.totalCost || "0").toFixed(2),
    format(new Date(e.createdAt), "MMM d, yyyy"),
  ].map(escape).join(","));

  const csv = [header, ...csvRows].join("\n");
  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
};
```

Note: `format` from date-fns and `projectSlug` are already available. `parseFloat` used in the existing render can be replaced with `Number()` for consistency — both work identically here.
