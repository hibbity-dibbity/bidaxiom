```typescript
const exportCsv = () => {
  const rows = table.getRowModel().rows;
  const today = new Date().toISOString().split("T")[0];
  const filename = `projects-${today}.csv`;

  const escape = (v: string) => `"${String(v).replace(/"/g, '""')}"`;

  const header = ["Job #", "Name", "Customer", "Phase", "Start Date"].map(escape).join(",");
  const csvRows = rows.map((row) => {
    const p = row.original;
    return [
      p.jobNumber ?? "",
      p.name,
      p.customerName ?? "",
      p.lifecyclePhase ?? "",
      p.startDate ? new Date(p.startDate).toLocaleDateString() : "",
    ].map(escape).join(",");
  });

  const csv = [header, ...csvRows].join("\n");
  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
};
```

```typescript
const exportCsv = () => {
  const rows = table.getRowModel().rows;
  const today = new Date().toISOString().split("T")[0];
  const filename = `subcontractors-${today}.csv`;

  const escape = (v: string) => `"${String(v).replace(/"/g, '""')}"`;

  const formatTrades = (trades: number[]) => {
    const names = trades.slice(0, 3).map((t) => {
      const div = CSI_DIVISIONS.find((d) => d.number === t);
      return div ? `${div.number} - ${div.name}` : String(t);
    });
    if (trades.length > 3) names.push(`+${trades.length - 3} more`);
    return names.join(" | ");
  };

  const header = ["Company", "Contact", "Email", "Trade", "Status"].map(escape).join(",");
  const csvRows = rows.map((row) => {
    const s = row.original;
    return [
      s.companyName ?? "",
      s.contactName ?? "",
      s.contactEmail ?? "",
      formatTrades(s.trades),
      getStatusLabel(s.status),
    ].map(escape).join(",");
  });

  const csv = [header, ...csvRows].join("\n");
  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
};
```
