**CSV export function** (place above the return statement):
```ts
const exportCsv = () => {
  const rows = table.getRowModel().rows;
  const today = new Date().toISOString().split("T")[0];
  const filename = `payment-apps-${projectSlug}-${today}.csv`;

  const escape = (v: string) => `"${String(v).replace(/"/g, '""')}"`;

  const header = ["App #", "Status", "Period Start", "Period End", "Amount Due"].map(escape).join(",");
  const csvRows = rows.map((row) => {
    const pa = row.original;
    return [
      pa.applicationNumber ?? "",
      pa.status.replace(/_/g, " "),
      new Date(pa.periodStart).toLocaleDateString(),
      new Date(pa.periodEnd).toLocaleDateString(),
      Number(pa.currentPaymentDue).toFixed(2),
    ].map(escape).join(",");
  });

  const csv = [header, ...csvRows].join("\n");
  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
};
```

const utils = trpc.useUtils();

const approveMutation = trpc.paymentApp.approve.useMutation({
  onSuccess: () => {
    utils.paymentApp.list.invalidate({ projectId });
    setPendingActionId(null);
    setPendingAction(null);
    setActionComments("");
  },
});

const markPaidMutation = trpc.paymentApp.markPaid.useMutation({

    setPendingAction(null);
    setActionComments("");
  },
});

**Handle confirm** (helper function):
```ts
const handleConfirm = () => {
  if (!pendingActionId || !pendingAction) return;
  if (pendingAction === "approve") {
    approveMutation.mutate({ paymentAppId: pendingActionId, comments: actionComments || undefined });
  } else {
    markPaidMutation.mutate({ paymentAppId: pendingActionId });
  }
};

const isMutating = approveMutation.isPending || markPaidMutation.isPending;
```
