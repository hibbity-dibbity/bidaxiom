```tsx
function DatePickerField({
  label,
  value,
  projectId,
  field,
  onSave,
}: {
  label: string; // "Set start date" | "Set end date"
  value: string | null | undefined;
  projectId: string;
  field: "startDate" | "endDate";
  onSave: (isoDate: string | null) => void;
}) {
  const [open, setOpen] = useState(false);

  const formatted = value
    ? new Date(value).toLocaleDateString("en-US", {
        month: "short",
        day: "numeric",
        year: "numeric",
      })
    : null;

  const selected = value ? new Date(value) : undefined;

  return (
    <Popover open={open} onOpenChange={setOpen}>
      <PopoverTrigger asChild>
        {formatted ? (
          <button className="text-sm text-muted-foreground hover:text-foreground transition-colors cursor-pointer">
            {formatted}
          </button>
        ) : (
          <button
            title={label}
            className="text-muted-foreground hover:text-foreground transition-colors cursor-pointer"
            aria-label={label}
          >
            <Calendar className="h-3.5 w-3.5" />
          </button>
        )}
      </PopoverTrigger>
      <PopoverContent className="w-auto p-0" align="start">
        <CalendarComponent
          mode="single"
          selected={selected}
          onSelect={(date) => {
            setOpen(false);
            onSave(date ? date.toISOString() : null);
          }}
          initialFocus
        />
      </PopoverContent>
    </Popover>
  );
}
```

**Helper functions to add** near the top of the file:

```tsx
function formatCurrencyDisplay(amount: number): { text: string; negative: boolean } {
  const negative = amount < 0;
  const abs = Math.abs(amount);
  const formatted = new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
    minimumFractionDigits: 0,
    maximumFractionDigits: 0,
  }).format(abs);
  return { text: negative ? `(${formatted})` : formatted, negative };
}

function formatDelta(delta: number | null): { text: string; over: boolean } | null {
  if (delta === null) return null;
  const abs = Math.abs(delta);
  const formatted = new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
    minimumFractionDigits: 0,
    maximumFractionDigits: 0,
  }).format(abs);
  // delta = contractValue - customerBudget
  // positive delta means contract > customer budget = "over"
  // negative delta means contract < customer budget = "under"
  return { text: `${formatted} ${delta > 0 ? "over" : "under"}`, over: delta > 0 };
}
```

IMPORTANT: Do NOT use `CurrencyDisplay` for the Profit KPI card. Use the local `formatCurrencyDisplay` helper defined above — it produces accounting parentheses format for negative values (e.g., `($1,200)`), which `CurrencyDisplay` does not support. All other KPI cards (Contract Value, Billed to Date, etc.) may continue using `CurrencyDisplay`.
