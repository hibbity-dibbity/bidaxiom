```typescript
  return (
    <Badge variant={config.variant}>

  );
```

```typescript
"use client";

import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import type { EstimateStatus } from "@construction/shared";
import { GenerateBidButton } from "../actions/GenerateBidButton";
import { SubmitEstimateButton } from "../actions/SubmitEstimateButton";
import { ConvertToProjectButton } from "../actions/ConvertToProjectButton";

interface NextStepsPanelProps {
  estimateId: string;
  status: EstimateStatus;
  projectId: string;
  onActionComplete?: () => void;
}

export function NextStepsPanel({ estimateId, status, projectId, onActionComplete }: NextStepsPanelProps) {
  const renderActions = () => {
    switch (status) {
      case "draft":
        return (
          <div className="space-y-3">
            <p className="text-sm text-muted-foreground">
              Your estimate is ready. Submit for approval or generate a bid to send to clients.
            </p>
            <div className="flex gap-2">
              <SubmitEstimateButton
                estimateId={estimateId}
                onSuccess={onActionComplete}
                variant="default"
              />
              <GenerateBidButton
                estimateId={estimateId}
                onSuccess={onActionComplete}
                variant="outline"
              />
            </div>
          </div>
        );

      case "submitted":
        return (
          <div className="space-y-3">
            <p className="text-sm text-muted-foreground">
              Estimate has been submitted. Mark as accepted if bid is won.
            </p>
            <div className="flex gap-2">
              <GenerateBidButton
                estimateId={estimateId}
                onSuccess={onActionComplete}
                variant="outline"
              />
            </div>
          </div>
        );

      case "accepted":
        return (
          <div className="space-y-3">
            <p className="text-sm text-muted-foreground">
              Congratulations! Convert this estimate to an active project with one click.
            </p>
            <div className="flex gap-2">
              <ConvertToProjectButton
                estimateId={estimateId}
                projectId={projectId}
                onSuccess={onActionComplete}
                variant="default"
              />
            </div>
          </div>
        );

      case "under_contract":
        return (
          <div className="space-y-3">
            <p className="text-sm text-muted-foreground">
              This estimate has been converted to an active project. Budget line items have been created.
            </p>
          </div>
        );

      default:
        return null;
    }
  };

  return (
    <Card>
      <CardHeader>
        <CardTitle>Next Steps</CardTitle>
      </CardHeader>
      <CardContent>
        {renderActions()}
      </CardContent>
    </Card>
  );
}
```

**1. Create apps/web/components/estimates/actions/GenerateBidButton.tsx:**

```typescript
"use client";

import { useState } from "react";
import { Button } from "@/components/ui/button";
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { trpc } from "@/lib/trpc/client";
import { useToast } from "@/hooks/use-toast";
import type { ButtonProps } from "@/components/ui/button";

interface GenerateBidButtonProps {
  estimateId: string;
  onSuccess?: () => void;
  variant?: ButtonProps["variant"];
}

export function GenerateBidButton({ estimateId, onSuccess, variant = "outline" }: GenerateBidButtonProps) {
  const [open, setOpen] = useState(false);
  const [bidName, setBidName] = useState("");
  const { toast } = useToast();

  const generateBid = trpc.estimates.generateBid.useMutation({
    onSuccess: () => {
      toast({
        title: "Bid generated",
        description: "Your bid has been created successfully.",
      });
      setOpen(false);
      setBidName("");
      onSuccess?.();
    },
    onError: (error) => {
      toast({
        title: "Error",
        description: error.message,
        variant: "destructive",
      });
    },
  });

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    generateBid.mutate({ estimateId, bidName });
  };

  return (
    <>
      <Button variant={variant} onClick={() => setOpen(true)}>
        Generate Bid
      </Button>

      <Dialog open={open} onOpenChange={setOpen}>
        <DialogContent>
          <form onSubmit={handleSubmit}>
            <DialogHeader>
              <DialogTitle>Generate Bid</DialogTitle>
              <DialogDescription>
                Create a bid from this estimate. You can submit the bid to clients after generation.
              </DialogDescription>
            </DialogHeader>

            <div className="space-y-4 py-4">
              <div className="space-y-2">
                <Label htmlFor="bidName">Bid Name</Label>
                <Input
                  id="bidName"
                  value={bidName}
                  onChange={(e) => setBidName(e.target.value)}
                  placeholder="e.g., ABC Construction Proposal"
                  required
                />
              </div>
            </div>

            <DialogFooter>
              <Button type="button" variant="outline" onClick={() => setOpen(false)}>
                Cancel
              </Button>
              <Button type="submit" disabled={generateBid.isPending || !bidName}>
                {generateBid.isPending ? "Generating..." : "Generate Bid"}
              </Button>
            </DialogFooter>
          </form>
        </DialogContent>
      </Dialog>
    </>
  );
}
```

```typescript
"use client";

import { useRouter } from "next/navigation";
import { Button } from "@/components/ui/button";
import { trpc } from "@/lib/trpc/client";
import { useToast } from "@/hooks/use-toast";
import type { ButtonProps } from "@/components/ui/button";

interface ConvertToProjectButtonProps {
  estimateId: string;
  projectId: string;
  onSuccess?: () => void;
  variant?: ButtonProps["variant"];
}

export function ConvertToProjectButton({
  estimateId,
  projectId,
  onSuccess,
  variant = "default"
}: ConvertToProjectButtonProps) {
  const { toast } = useToast();
  const router = useRouter();

  const convertToProject = trpc.estimates.convertToProject.useMutation({
    onSuccess: (data) => {
      toast({
        title: "Project activated",
        description: `Budget created with ${data.budgetLineItemsCreated} line items.`,
      });
      onSuccess?.();
      // Navigate to project budget page
      router.push(`/projects/${projectId}/budget`);
    },
    onError: (error) => {
      toast({
        title: "Error",
        description: error.message,
        variant: "destructive",
      });
    },
  });

  return (
    <Button
      variant={variant}
      onClick={() => convertToProject.mutate({ estimateId })}
      disabled={convertToProject.isPending}
    >
      {convertToProject.isPending ? "Converting..." : "Convert to Project"}
    </Button>
  );
}
```
