Create generateBidFromEstimate function following research Pattern 2 (Immutable Bid Snapshots):
```typescript
export async function generateBidFromEstimate(
  ctx: TenantContext,
  input: { estimateId: string; bidName: string; userId: string }
) {
  const { db, schema } = ctx.tenant;

  // 1. Get estimate with line items
  const estimate = await db.query.estimates.findFirst({
    where: eq(schema.estimates.id, input.estimateId),
    with: {
      lineItems: true,
    },
  });

  if (!estimate) {
    throw new TRPCError({
      code: "NOT_FOUND",
      message: "Estimate not found",
    });
  }

  if (estimate.status !== "draft" && estimate.status !== "submitted") {
    throw new TRPCError({
      code: "BAD_REQUEST",
      message: "Can only generate bid from draft or submitted estimates",
    });
  }

  // 2. Create bid with immutable snapshot in transaction
  return await db.transaction(async (tx) => {
    const [bid] = await tx.insert(schema.bids).values({
      estimateId: input.estimateId,
      projectId: estimate.projectId,
      rfqId: null, // Estimate-based bids don't have RFQ
      invitationId: null,
      subcontractorId: null,
      bidderCompanyName: input.bidName,
      bidderContactEmail: null,
      currentVersionId: null,
      status: "draft",
      notes: `Generated from estimate: ${estimate.name}`,
    }).returning();

    // 3. Create bid version with snapshot
    const [bidVersion] = await tx.insert(schema.bidVersions).values({
      bidId: bid.id,
      versionNumber: 1,
      totalAmount: estimate.totalCost,
      lineItems: estimate.lineItems.map(item => ({
        description: item.description,
        quantity: Number(item.quantity),
        unitPrice: Number(item.unitCost),
        amount: Number(item.unitCost) * Number(item.quantity),
      })),
      scopeNotes: `Estimate breakdown:\nDirect Costs: ${estimate.directCosts}\nOverhead: ${estimate.overheadAmount}\nMarkup: ${estimate.markupAmount}\nTotal: ${estimate.totalCost}`,
      createdBy: input.userId,
    }).returning();

    // 4. Update bid with current version
    await tx.update(schema.bids)
      .set({ currentVersionId: bidVersion.id })
      .where(eq(schema.bids.id, bid.id));

    // 5. Link estimate to bid
    await tx.update(schema.estimates)
      .set({ linkedBidId: bid.id })
      .where(eq(schema.estimates.id, input.estimateId));

    return bid;
  });
}
```

submitEstimate: protectedProcedure
  .input(z.object({ estimateId: z.string().uuid() }))
  .mutation(async ({ ctx, input }) => {
    const { db, schema } = ctx.tenant;

    const estimate = await db.query.estimates.findFirst({
      where: eq(schema.estimates.id, input.estimateId),
    });

    if (!estimate) {
      throw new TRPCError({ code: "NOT_FOUND", message: "Estimate not found" });
    }

    if (!canTransitionEstimate(estimate.status as EstimateStatus, "submitted")) {
      throw new TRPCError({
        code: "BAD_REQUEST",
        message: `Cannot submit estimate with status '${estimate.status}'`,
      });
    }

    return await db.transaction(async (tx) => {
      // Update estimate status
      const [updated] = await tx.update(schema.estimates)
        .set({
          status: "submitted",
          submittedAt: new Date(),
        })
        .where(eq(schema.estimates.id, input.estimateId))
        .returning();

      // Update project status to "bidding"
      await tx.update(schema.projects)
        .set({ status: "bidding" })
        .where(eq(schema.projects.id, estimate.projectId));

      return updated;
    });
  }),
```

updateBidStatus: protectedProcedure
  .input(updateBidStatusSchema)
  .mutation(async ({ ctx, input }) => {
    const { db, schema } = ctx.tenant;

    const bid = await db.query.bids.findFirst({
      where: eq(schema.bids.id, input.bidId),
    });

    if (!bid) {
      throw new TRPCError({ code: "NOT_FOUND", message: "Bid not found" });
    }

    if (bid.status !== "submitted" && (input.status === "won" || input.status === "lost")) {
      throw new TRPCError({
        code: "BAD_REQUEST",
        message: "Can only mark submitted bids as won/lost",
      });
    }

    return await db.transaction(async (tx) => {
      // Update bid status
      const [updated] = await tx.update(schema.bids)
        .set({
          status: input.status,
          closedAt: input.status === "won" || input.status === "lost" ? new Date() : undefined,
          closedBy: input.status === "won" || input.status === "lost" ? ctx.auth.userId : undefined,
          closedNotes: input.notes,
        })
        .where(eq(schema.bids.id, input.bidId))
        .returning();

      // If won, update estimate to accepted
      if (input.status === "won" && bid.estimateId) {
        await tx.update(schema.estimates)
          .set({
            status: "accepted",
            acceptedAt: new Date(),
          })
          .where(eq(schema.estimates.id, bid.estimateId));
      }

      return updated;
    });
  }),
```
