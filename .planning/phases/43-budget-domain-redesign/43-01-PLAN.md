```typescript
import { useState, useCallback } from "react";
import type { VisibilityState, Updater } from "@tanstack/react-table";

export function useColumnVisibility(tableKey: string) {
  const storageKey = `col-picker:${tableKey}`;

  const [columnVisibility, setColumnVisibility] = useState<VisibilityState>(() => {
    if (typeof window === "undefined") return {};
    try {
      const stored = localStorage.getItem(storageKey);
      return stored ? (JSON.parse(stored) as VisibilityState) : {};
    } catch {
      return {};
    }
  });

  const onColumnVisibilityChange = useCallback(
    (updater: Updater<VisibilityState>) => {
      setColumnVisibility((prev) => {
        const next = typeof updater === "function" ? updater(prev) : updater;
        try {
          localStorage.setItem(storageKey, JSON.stringify(next));
        } catch {
          /* storage unavailable — degrade silently */
        }
        return next;
      });
    },
    [storageKey]
  );

  return { columnVisibility, onColumnVisibilityChange };
}
```

Key requirements:
- Lazy `useState` initializer with `typeof window === "undefined"` guard prevents SSR hydration mismatch
- `useCallback` with `[storageKey]` dependency ensures stable callback reference — prevents infinite re-render loop (Pitfall 5 from research)
- The `storageKey` uses table-scoped prefix `col-picker:${tableKey}` to prevent cross-table key collision (Pitfall 4 from research)
- localStorage write uses functional `setColumnVisibility` form so it always has the latest state

**File 1: column-picker.tsx**

```typescript
"use client";

import { Settings2 } from "lucide-react";
import type { Table } from "@tanstack/react-table";
import { Button } from "@/components/ui/button";
import { Checkbox } from "@/components/ui/checkbox";
import {
  Popover,
  PopoverContent,
  PopoverTrigger,
} from "@/components/ui/popover";

interface ColumnPickerProps<TData> {
  table: Table<TData>;
}

export function ColumnPicker<TData>({ table }: ColumnPickerProps<TData>) {
  const hideableColumns = table
    .getAllLeafColumns()
    .filter((col) => col.getCanHide());

  if (hideableColumns.length === 0) return null;

  return (
    <Popover>
      <PopoverTrigger asChild>
        <Button
          variant="ghost"
          size="sm"
          className="h-8 w-8 p-0"
          title="Column settings"
        >
          <Settings2 className="h-4 w-4" />
          <span className="sr-only">Column settings</span>
        </Button>
      </PopoverTrigger>
      <PopoverContent align="end" className="w-48 p-2">
        <p className="text-xs font-medium text-muted-foreground px-2 pb-2">
          Toggle columns
        </p>
        <div className="space-y-1">
          {hideableColumns.map((col) => {
            const label =
              typeof col.columnDef.header === "string"
                ? col.columnDef.header
                : col.id;
            return (
              <label
                key={col.id}
                className="flex items-center gap-2 px-2 py-1.5 text-sm cursor-pointer hover:bg-muted rounded"
              >
                <Checkbox
                  checked={col.getIsVisible()}
                  onCheckedChange={(checked) =>
                    col.toggleVisibility(!!checked)
                  }
                />
                {label}
              </label>
            );
          })}
        </div>
      </PopoverContent>
    </Popover>
  );
}
```

Key requirements:
- Uses `col.getCanHide()` to filter — never shows pinned columns (`enableHiding: false`) in the list (research Pitfall: do NOT call `getAllLeafColumns()` without this filter)
- `align="end"` so the popover opens to the left of the gear icon (top-right placement)
- Returns `null` if no hideable columns — safe to render in any context

```typescript
"use client";

import { X } from "lucide-react";
import { Badge } from "@/components/ui/badge";

export interface FilterChip {
  label: string;
  onRemove: () => void;
}

interface FilterChipsProps {
  chips: FilterChip[];
  onClearAll: () => void;
}

export function FilterChips({ chips, onClearAll }: FilterChipsProps) {
  if (chips.length === 0) return null;

  return (
    <div className="flex flex-wrap items-center gap-2">
      {chips.map((chip, i) => (
        <Badge key={i} variant="secondary" className="gap-1 pr-1 h-6">
          <span className="text-xs">{chip.label}</span>
          <button
            onClick={chip.onRemove}
            className="ml-1 rounded-full hover:bg-muted-foreground/20 p-0.5 transition-colors"
            type="button"
          >
            <X className="h-3 w-3" />
            <span className="sr-only">Remove {chip.label} filter</span>
          </button>
        </Badge>
      ))}
      <button
        onClick={onClearAll}
        className="text-xs text-muted-foreground hover:text-foreground underline transition-colors"
        type="button"
      >
        Clear all
      </button>
    </div>
  );
}
```
