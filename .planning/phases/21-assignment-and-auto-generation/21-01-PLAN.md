   // Get company profile for job number settings
   const [companyProfile] = await db
     .select()
     .from(schema.companyProfiles)
     .where(isNull(schema.companyProfiles.validTo))
     .limit(1);

2. **Auto-generation logic (when no manual job number provided):**
   ```typescript
   let finalJobNumber = input.jobNumber; // Start with manual input (if provided)
   let updatedJobSettings = jobNumberSettings;

   if (!finalJobNumber && jobNumberSettings?.enabled) {
     // Auto-generate job number
     const result = generateJobNumber(jobNumberSettings);
     finalJobNumber = result.jobNumber;
     updatedJobSettings = result.updatedSettings;
   }
   ```

3. **Manual counter increment logic (when manual job number provided):**
   ```typescript
   // If manual job number provided and auto-generation is enabled, increment counter
   // to prevent future auto-generated numbers from conflicting
   if (input.jobNumber && jobNumberSettings?.enabled) {
     // Extract sequence number from manual job number
     // Example: "26-1234" with template "{YY}-{NNNN}" -> sequence is 1234
     const sequenceMatch = input.jobNumber.match(/(\d+)$/);
     if (sequenceMatch) {
       const manualSequence = parseInt(sequenceMatch[1], 10);
       // Set nextSequenceNumber to be one higher than manual entry
       if (manualSequence >= jobNumberSettings.nextSequenceNumber) {
         updatedJobSettings = {
           ...jobNumberSettings,
           nextSequenceNumber: manualSequence + 1,
         };
       }
     }
   }
   ```

4. **Update company profile with new counter (inside transaction):**
   ```typescript
   const project = await db.transaction(async (tx) => {
     // Update company profile with new sequence number if changed
     if (updatedJobSettings && updatedJobSettings !== jobNumberSettings) {
       await tx
         .update(schema.companyProfiles)
         .set({ jobNumberSettings: updatedJobSettings })
         .where(isNull(schema.companyProfiles.validTo));
     }

     // Insert project with finalJobNumber
     const [newProject] = await tx
       .insert(schema.projects)
       .values({
         name: input.name,
         jobNumber: finalJobNumber || null, // Use auto-generated or manual
         slug: projectSlug,
         // ... rest of fields
       })
       .returning();

     // ... rest of transaction (team, folders, phases)
   });
   ```

**Why this approach:**
- Import generateJobNumber at top: `import { generateJobNumber } from "../../utils/job-number.js";`
- Fetch settings before transaction for cleaner code
- Manual counter increment prevents duplicates when user enters "26-1234" and next auto would be "26-1233"
- Update happens in same transaction as project creation for atomicity
- Uses existing `isNull(validTo)` pattern to update active company profile

**What to avoid:**
- Don't create a separate mutation — keep it in project.create for atomicity
- Don't increment counter after transaction — must be atomic
- Don't skip counter increment for manual entries — this is CRITICAL to prevent duplicates (JOBASGN-03)

**Implementation:**

```typescript
/**
 * Check if a job number is already in use.
 * Used for duplicate validation before project creation.
 */
checkJobNumberAvailability: protectedProcedure
  .input(z.object({ jobNumber: z.string().min(1) }))
  .query(async ({ ctx, input }) => {
    if (!ctx.tenant) {
      throw new TRPCError({
        code: "INTERNAL_SERVER_ERROR",
        message: "Tenant context missing",
      });
    }

    const [existing] = await ctx.tenant.db
      .select({ id: ctx.tenant.schema.projects.id })
      .from(ctx.tenant.schema.projects)
      .where(eq(ctx.tenant.schema.projects.jobNumber, input.jobNumber))
      .limit(1);

    return {
      available: !existing,
      existingProjectId: existing?.id || null,
    };
  }),
```

**Why this approach:**
- Simple query — just check if job number exists
- Returns both availability flag and existing project ID (useful for error messages)
- Will be called from frontend in Phase 22 for real-time validation

**What to avoid:**
- Don't throw error — return availability status instead
- Don't cache results — job numbers can be created by other users
