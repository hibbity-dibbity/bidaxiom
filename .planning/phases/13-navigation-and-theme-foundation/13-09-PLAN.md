```typescript
import { db } from "./db/pool";
import { sql } from "drizzle-orm";

async function migrateLogo() {
  console.log("Adding logoStorageKey to company_profiles...");

  // Get all tenant schemas
  const schemasResult = await db.execute(sql`
    SELECT schema_name
    FROM information_schema.schemata
    WHERE schema_name LIKE 'tenant_%'
  `);

  const schemas = schemasResult.rows.map((r: any) => r.schema_name);
  console.log(`Found ${schemas.length} tenant schemas`);

  for (const schema of schemas) {
    console.log(`Migrating ${schema}...`);

    // Check if column exists
    const checkResult = await db.execute(sql`
      SELECT column_name
      FROM information_schema.columns
      WHERE table_schema = ${schema}
      AND table_name = 'company_profiles'
      AND column_name = 'logo_storage_key'
    `);

    if (checkResult.rows.length === 0) {
      // Add column
      await db.execute(sql`
        ALTER TABLE ${sql.raw(schema)}.company_profiles
        ADD COLUMN IF NOT EXISTS logo_storage_key TEXT
      `);
      console.log(`  ✓ Added logo_storage_key to ${schema}`);
    } else {
      console.log(`  - Column already exists in ${schema}`);
    }
  }

  console.log("Migration complete!");
  process.exit(0);
}

migrateLogo().catch(console.error);
```
