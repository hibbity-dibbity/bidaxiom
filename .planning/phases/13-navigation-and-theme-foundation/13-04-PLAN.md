```typescript
"use client";

import { useQueryState } from "nuqs";

/**
 * URL-based navigation state management using nuqs.
 * Enables shareable/bookmarkable navigation states.
 *
 * Example: /projects?view=grid&status=active
 */

export function useNavigationState() {
  const [view, setView] = useQueryState("view", {
    defaultValue: "list",
    shallow: true, // Don't trigger navigation
  });

  const [status, setStatus] = useQueryState("status", {
    defaultValue: "all",
    shallow: true,
  });

  return {
    view,
    setView,
    status,
    setStatus,
  };
}
```

```typescript
"use client";

import { useEffect, useState } from "react";
import {
  Sidebar as ShadcnSidebar,
  SidebarContent,
  SidebarHeader,
  SidebarFooter,
  SidebarGroup,
  SidebarGroupLabel,
  useSidebar,
} from "@/components/ui/sidebar";
import { SidebarNav } from "./sidebar-nav";

const STORAGE_KEY = "sidebar-collapsed";

export function AppSidebar() {
  const { setOpen, open } = useSidebar();
  const [mounted, setMounted] = useState(false);

  // Restore collapse state from localStorage on mount
  useEffect(() => {
    setMounted(true);
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored !== null) {
      setOpen(stored === "true");
    }
  }, [setOpen]);

  // Persist collapse state changes
  useEffect(() => {
    if (mounted) {
      localStorage.setItem(STORAGE_KEY, String(open));
    }
  }, [open, mounted]);

  return (
    <ShadcnSidebar collapsible="icon">
      <SidebarHeader className="border-b px-2 py-3">
        <div className="flex items-center gap-2 px-2">
          <div className="flex h-8 w-8 items-center justify-center rounded-md bg-primary text-primary-foreground font-bold text-sm">
            CS
          </div>
          <span className="font-semibold text-sm group-data-[collapsible=icon]:hidden">
            ConstructionSaaS
          </span>
        </div>
      </SidebarHeader>

      <SidebarContent>
        <SidebarGroup>
          <SidebarGroupLabel>Navigation</SidebarGroupLabel>
          <SidebarNav />
        </SidebarGroup>
      </SidebarContent>

      <SidebarFooter className="border-t p-2">
        <div className="text-xs text-muted-foreground px-2 group-data-[collapsible=icon]:hidden">
          v0.0.1
        </div>
      </SidebarFooter>
    </ShadcnSidebar>
  );
}
```

Why localStorage (not cookie):
- Sidebar state is client-side preference only
- Doesn't affect SSR or need server awareness
- No FOUC concern (visual preference, not theme)
- Faster to read/write than cookie parsing

Hydration handling:
- mounted state prevents SSR mismatch
- Effect runs after first render
- Gracefully handles localStorage unavailable (SSR, private browsing)

```typescript
"use client";

import { usePathname } from "next/navigation";
import Link from "next/link";
import {
  Home,
  FolderKanban,
  DollarSign,
  FileText,
  Users,
  HardHat,
  Settings,
  BarChart,
  Briefcase,
} from "lucide-react";
import {
  SidebarMenu,
  SidebarMenuItem,
  SidebarMenuButton,
} from "@/components/ui/sidebar";

const mainNav = [
  { href: "/", label: "Dashboard", icon: Home },
  { href: "/projects", label: "Projects", icon: FolderKanban },
  { href: "/customers", label: "Customers", icon: Users },
  { href: "/subcontractors", label: "Subcontractors", icon: HardHat },
  { href: "/bids", label: "Bidding", icon: Briefcase },
  { href: "/financials", label: "Financials", icon: DollarSign },
  { href: "/documents", label: "Documents", icon: FileText },
  { href: "/reports", label: "Reports", icon: BarChart },
  { href: "/settings", label: "Settings", icon: Settings },
];

// Contextual nav items shown when viewing a project
const projectNav = [
  { href: "/overview", label: "Overview", icon: Home },
  { href: "/budget", label: "Budget", icon: DollarSign },
  { href: "/change-orders", label: "Change Orders", icon: FileText },
  { href: "/documents", label: "Documents", icon: FileText },
];

export function SidebarNav() {
  const pathname = usePathname();

  // Detect if we're viewing a project detail page
  const isProjectPage = pathname?.startsWith("/projects/") && pathname !== "/projects";

  // Extract project ID from URL if on project page
  const projectId = isProjectPage ? pathname?.split("/")[2] : null;

  // Use contextual nav when viewing project, main nav otherwise
  const navItems = isProjectPage
    ? projectNav.map((item) => ({
        ...item,
        href: `/projects/${projectId}${item.href}`,
      }))
    : mainNav;

  return (
    <SidebarMenu>
      {navItems.map((item) => {
        const Icon = item.icon;
        const isActive = pathname === item.href;

        return (
          <SidebarMenuItem key={item.href}>
            <SidebarMenuButton asChild isActive={isActive}>
              <Link href={item.href}>
                <Icon className="h-4 w-4" />
                <span>{item.label}</span>
              </Link>
            </SidebarMenuButton>
          </SidebarMenuItem>
        );
      })}
    </SidebarMenu>
  );
}
```
