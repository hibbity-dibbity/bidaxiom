```typescript
export async function generateLogoUploadUrl(
  tenantId: string,
  fileExtension: string // e.g., "png", "jpg", "svg"
): Promise<{ uploadUrl: string; storageKey: string }> {
  const storageKey = `tenants/${tenantId}/branding/logo.${fileExtension}`;
  const uploadUrl = await s3Client.presignedPutObject(
    process.env.S3_BUCKET_NAME!,
    storageKey,
    15 * 60 // 15 minutes expiry
  );
  return { uploadUrl, storageKey };
}

export async function generateLogoDownloadUrl(
  storageKey: string
): Promise<string> {
  return s3Client.presignedGetObject(
    process.env.S3_BUCKET_NAME!,
    storageKey,
    60 * 60 // 1 hour expiry
  );
}

export async function deleteLogoFromStorage(
  storageKey: string
): Promise<void> {
  await s3Client.removeObject(
    process.env.S3_BUCKET_NAME!,
    storageKey
  );
}
```

**apps/api/src/routers/company.ts** (UPDATE):
- Import storage service helpers
- Add three new procedures (use authorizedProcedure for CASL check):

```typescript
uploadLogo: authorizedProcedure
  .input(
    z.object({
      fileExtension: z.enum(["png", "jpg", "jpeg", "svg", "webp"]),
    })
  )
  .mutation(async ({ ctx, input }) => {
    // CASL check: must be org:admin to upload logo
    if (!ctx.abilities.can("update", "CompanyProfile")) {
      throw new TRPCError({ code: "FORBIDDEN" });
    }

    const { uploadUrl, storageKey } = await generateLogoUploadUrl(
      ctx.tenantId,
      input.fileExtension
    );

    // Update company profile with new storage key
    await ctx.db
      .update(companyProfiles)
      .set({
        logoStorageKey: storageKey,
        updatedAt: new Date()
      })
      .where(
        and(
          eq(companyProfiles.validTo, null) // current version only
        )
      );

    return { uploadUrl, storageKey };
  }),

getLogo: authorizedProcedure
  .query(async ({ ctx }) => {
    // Anyone in org can view logo
    const profile = await ctx.db.query.companyProfiles.findFirst({
      where: eq(companyProfiles.validTo, null),
      columns: { logoStorageKey: true },
    });

    if (!profile?.logoStorageKey) {
      return { logoUrl: null };
    }

    const logoUrl = await generateLogoDownloadUrl(profile.logoStorageKey);
    return { logoUrl };
  }),

deleteLogo: authorizedProcedure
  .mutation(async ({ ctx }) => {
    // CASL check: must be org:admin to delete logo
    if (!ctx.abilities.can("update", "CompanyProfile")) {
      throw new TRPCError({ code: "FORBIDDEN" });
    }

    const profile = await ctx.db.query.companyProfiles.findFirst({
      where: eq(companyProfiles.validTo, null),
      columns: { logoStorageKey: true },
    });

    if (profile?.logoStorageKey) {
      // Delete from S3
      await deleteLogoFromStorage(profile.logoStorageKey);

      // Clear storage key in database
      await ctx.db
        .update(companyProfiles)
        .set({
          logoStorageKey: null,
          updatedAt: new Date()
        })
        .where(eq(companyProfiles.validTo, null));
    }

    return { success: true };
  }),
```
