```typescript
"use client";

import { useState, useEffect } from "react";
import Image from "next/image";
import { Building2 } from "lucide-react";
import { trpc } from "@/lib/trpc";
import { cn } from "@/lib/utils";

interface CompanyLogoProps {
  /**
   * Display variant:
   * - sidebar: Sidebar header (icon + text or icon-only when collapsed)
   * - topbar: Top navigation bar (compact)
   * - auth: Sign-in page (large, centered)
   */
  variant: "sidebar" | "topbar" | "auth";
  /**
   * Optional className for custom styling
   */
  className?: string;
  /**
   * Show company name alongside logo (if space permits)
   */
  showName?: boolean;
}

export function CompanyLogo({
  variant,
  className,
  showName = true,
}: CompanyLogoProps) {
  const { data, isLoading } = trpc.company.getLogo.useQuery(undefined, {
    staleTime: 5 * 60 * 1000, // Cache for 5 minutes
    refetchOnWindowFocus: false,
  });

  const [imageError, setImageError] = useState(false);
  const logoUrl = data?.logoUrl;
  const hasLogo = logoUrl && !imageError;

  // Reset error state when logo URL changes
  useEffect(() => {
    setImageError(false);
  }, [logoUrl]);

  // Variant-specific sizing
  const sizes = {
    sidebar: { icon: 32, container: "h-8 w-8" },
    topbar: { icon: 28, container: "h-7 w-7" },
    auth: { icon: 64, container: "h-16 w-16" },
  };

  const size = sizes[variant];

  return (
    <div
      className={cn(
        "flex items-center gap-2",
        variant === "auth" && "flex-col",
        className
      )}
    >
      {/* Logo image or fallback icon */}
      <div
        className={cn(
          "flex items-center justify-center rounded-md overflow-hidden",
          size.container,
          !hasLogo && "bg-construction text-construction-foreground"
        )}
      >
        {hasLogo ? (
          <Image
            src={logoUrl}
            alt="Company logo"
            width={size.icon}
            height={size.icon}
            className="object-contain"
            onError={() => setImageError(true)}
            unoptimized // S3 pre-signed URLs don't work with Next.js image optimization
          />
        ) : (
          <Building2 className={`h-${size.icon / 4} w-${size.icon / 4}`} />
        )}
      </div>

      {/* Company name (conditionally shown based on variant and prop) */}
      {showName && (
        <span
          className={cn(
            "font-semibold",
            variant === "sidebar" && "text-sm group-data-[collapsible=icon]:hidden",
            variant === "topbar" && "text-sm",
            variant === "auth" && "text-xl"
          )}
        >
          ConstructionSaaS
        </span>
      )}
    </div>
  );
}
```

- Supports three variants: sidebar, topbar, auth

- Responsive to sidebar collapse state (group-data selector)

```typescript
"use client";

import { useState, useRef } from "react";
import { Upload, Trash2, Loader2 } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { CompanyLogo } from "./company-logo";
import { trpc } from "@/lib/trpc";
import { toast } from "sonner";

const ACCEPTED_FORMATS = ["image/png", "image/jpeg", "image/jpg", "image/svg+xml", "image/webp"];
const MAX_FILE_SIZE = 2 * 1024 * 1024; // 2MB

export function LogoUpload() {
  const utils = trpc.useUtils();
  const fileInputRef = useRef<HTMLInputElement>(null);
  const [uploading, setUploading] = useState(false);

  const { data: currentLogo } = trpc.company.getLogo.useQuery();

  const uploadLogoMutation = trpc.company.uploadLogo.useMutation();
  const deleteLogoMutation = trpc.company.deleteLogo.useMutation({
    onSuccess: () => {
      toast.success("Logo deleted successfully");
      utils.company.getLogo.invalidate();
    },
    onError: (error) => {
      toast.error(`Failed to delete logo: ${error.message}`);
    },
  });

  const handleFileSelect = async (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (!file) return;

    // Validate file type
    if (!ACCEPTED_FORMATS.includes(file.type)) {
      toast.error("Invalid file format. Please upload PNG, JPG, SVG, or WebP.");
      return;
    }

    // Validate file size
    if (file.size > MAX_FILE_SIZE) {
      toast.error("File too large. Maximum size is 2MB.");
      return;
    }

    setUploading(true);

    try {
      // Get extension from file type
      const extension = file.type.split("/")[1];

      // Get pre-signed upload URL
      const { uploadUrl, storageKey } = await uploadLogoMutation.mutateAsync({
        fileExtension: extension as "png" | "jpg" | "jpeg" | "svg" | "webp",
      });

      // Upload file to S3
      const uploadResponse = await fetch(uploadUrl, {
        method: "PUT",
        body: file,
        headers: {
          "Content-Type": file.type,
        },
      });

      if (!uploadResponse.ok) {
        throw new Error("Upload failed");
      }

      toast.success("Logo uploaded successfully");
      utils.company.getLogo.invalidate(); // Refresh logo display
    } catch (error) {
      toast.error(`Failed to upload logo: ${error instanceof Error ? error.message : "Unknown error"}`);
    } finally {
      setUploading(false);
      if (fileInputRef.current) {
        fileInputRef.current.value = ""; // Reset input
      }
    }
  };

  return (
    <Card>
      <CardHeader>
        <CardTitle>Company Logo</CardTitle>
        <CardDescription>
          Upload your company logo. Recommended size: 512x512px. Max file size: 2MB.
        </CardDescription>
      </CardHeader>
      <CardContent className="space-y-4">
        {/* Current logo preview */}
        <div className="flex items-center gap-4">
          <CompanyLogo variant="auth" showName={false} />
          <div className="flex-1">
            {currentLogo?.logoUrl ? (
              <p className="text-sm text-muted-foreground">Current logo</p>
            ) : (
              <p className="text-sm text-muted-foreground">No logo uploaded</p>
            )}
          </div>
        </div>

        {/* Upload/Delete buttons */}
        <div className="flex gap-2">
          <Button
            onClick={() => fileInputRef.current?.click()}
            disabled={uploading}
            variant="outline"
          >
            {uploading ? (
              <>
                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
                Uploading...
              </>
            ) : (
              <>
                <Upload className="mr-2 h-4 w-4" />
                {currentLogo?.logoUrl ? "Replace Logo" : "Upload Logo"}
              </>
            )}
          </Button>

          {currentLogo?.logoUrl && (
            <Button
              onClick={() => deleteLogoMutation.mutate()}
              disabled={deleteLogoMutation.isLoading}
              variant="destructive"
            >
              {deleteLogoMutation.isLoading ? (
                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
              ) : (
                <Trash2 className="mr-2 h-4 w-4" />
              )}
              Delete Logo
            </Button>
          )}
        </div>

        {/* Hidden file input */}
        <input
          ref={fileInputRef}
          type="file"
          accept={ACCEPTED_FORMATS.join(",")}
          className="hidden"
          onChange={handleFileSelect}
        />
      </CardContent>
    </Card>
  );
}
```
