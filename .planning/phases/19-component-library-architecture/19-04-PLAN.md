```typescript
import slug from "slug";

/**
 * Generate URL-safe slug from identifier and name
 * Examples:
 *   - Project: "24001" + "Main Street Renovation" → "24001-main-street-renovation"
 *   - Customer: "CUST-123" + "ABC Construction" → "cust-123-abc-construction"
 *   - Estimate: "EST-24001" + "Main Street Estimate" → "est-24001-main-street-estimate"
 */
export function generateSlug(identifier: string, name: string): string {
  const slugIdentifier = slug(identifier, { lower: true });
  const slugName = slug(name, { lower: true });
  return `${slugIdentifier}-${slugName}`;
}

/**
 * Parse slug back to identifier
 * Examples:
 *   - "24001-main-street-renovation" → "24001"
 *   - "cust-123-abc-construction" → "CUST-123" (requires lookup)
 *   - "est-24001-main-street-estimate" → "EST-24001" (requires lookup)
 *
 * Note: parseSlug returns slug-safe identifier, not original.
 * Use for database lookup, not display.
 */
export function parseSlug(fullSlug: string): string {
  // Split on first dash after identifier
  // Identifier is typically numeric (projects) or prefix-numeric (customers/estimates)
  const parts = fullSlug.split("-");

  // Project pattern: "24001-..." (numeric identifier)
  if (/^\d+$/.test(parts[0])) {
    return parts[0];
  }

  // Customer/Estimate pattern: "cust-123-..." or "est-24001-..."
  if (parts.length >= 2 && /^[a-z]+$/.test(parts[0]) && /^\d+$/.test(parts[1])) {
    return `${parts[0]}-${parts[1]}`;
  }

  // Fallback: return first part
  return parts[0];
}

/**
 * Validate slug format
 */
export function isValidSlug(slug: string): boolean {
  return /^[a-z0-9]+(-[a-z0-9]+)*$/.test(slug);
}
```

Key improvements:
- Generic generateSlug() works for any entity type
- parseSlug() handles project (numeric) and prefixed (cust-123) identifiers
- Validation function for security (prevent path traversal)
- Documentation with examples for each entity type

async function addEntitySlugs() {
  const pool = new Pool({
    connectionString: process.env.DATABASE_URL,
  });

  const db = drizzle(pool);

  // Get all tenant schemas
  const tenants = await pool.query(`
    SELECT schema_name FROM information_schema.schemata
    WHERE schema_name LIKE 'tenant_%'
  `);

  for (const tenant of tenants.rows) {
    const schema = tenant.schema_name;
    console.log(`Processing ${schema}...`);

    // Add slug column to customers (if not exists)
    await pool.query(`
      ALTER TABLE ${schema}.customers
      ADD COLUMN IF NOT EXISTS slug TEXT UNIQUE
    `);

    // Add slug column to subcontractors (if not exists)
    await pool.query(`
      ALTER TABLE ${schema}.subcontractors
      ADD COLUMN IF NOT EXISTS slug TEXT UNIQUE
    `);

    // Add slug column to estimates (if not exists)
    await pool.query(`
      ALTER TABLE ${schema}.estimates
      ADD COLUMN IF NOT EXISTS slug TEXT UNIQUE
    `);

    // Generate slugs for existing customers
    const customers = await pool.query(`
      SELECT id, name FROM ${schema}.customers WHERE slug IS NULL
    `);
    for (const customer of customers.rows) {
      const slug = generateSlug(`cust-${customer.id.slice(0, 8)}`, customer.name);
      await pool.query(
        `UPDATE ${schema}.customers SET slug = $1 WHERE id = $2`,
        [slug, customer.id]
      );
    }

    // Generate slugs for existing subcontractors
    const subcontractors = await pool.query(`
      SELECT id, name FROM ${schema}.subcontractors WHERE slug IS NULL
    `);
    for (const subcontractor of subcontractors.rows) {
      const slug = generateSlug(`sub-${subcontractor.id.slice(0, 8)}`, subcontractor.name);
      await pool.query(
        `UPDATE ${schema}.subcontractors SET slug = $1 WHERE id = $2`,
        [slug, subcontractor.id]
      );
    }

    // Generate slugs for existing estimates
    const estimates = await pool.query(`
      SELECT e.id, e.name, p.job_number
      FROM ${schema}.estimates e
      JOIN ${schema}.projects p ON e.project_id = p.id
      WHERE e.slug IS NULL
    `);
    for (const estimate of estimates.rows) {
      const slug = generateSlug(`est-${estimate.job_number}`, estimate.name);
      await pool.query(
        `UPDATE ${schema}.estimates SET slug = $1 WHERE id = $2`,
        [slug, estimate.id]
      );
    }

    console.log(`✓ ${schema} complete`);
  }

  await pool.end();
  console.log("All tenants updated");
}

addEntitySlugs().catch(console.error);

export const customerRouter = router({

  getBySlug: publicProcedure
    .input(z.object({ slug: z.string() }))
    .query(async ({ ctx, input }) => {
      if (!isValidSlug(input.slug)) {
        throw new TRPCError({ code: "BAD_REQUEST", message: "Invalid slug format" });
      }

      const customer = await ctx.db
        .select()
        .from(schema.customers)
        .where(eq(schema.customers.slug, input.slug))
        .limit(1);

      if (!customer[0]) {
        throw new TRPCError({ code: "NOT_FOUND", message: "Customer not found" });
      }

      return customer[0];
    }),
});
