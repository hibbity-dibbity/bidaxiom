**FORM STRUCTURE (top to bottom) — replace the entire form JSX:**

```tsx
<Form {...form}>
  <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-6">

    {/* PROJECT NUMBER — inline widget, only shown when jobNumberSettings exists */}
    {jobNumberSettings && (
      <div className="space-y-1.5">
        <FormLabel>Project Number</FormLabel>
        {!manualJobNumber && jobNumberSettings.enabled ? (
          /* READ-ONLY MODE */
          <div className="flex items-center gap-3">
            <span className="font-mono text-base">
              {previewJobNumber(jobNumberSettings.template, jobNumberSettings.nextSequenceNumber)}
            </span>
            <Button
              type="button"
              variant="outline"
              size="sm"
              onClick={() => setManualJobNumber(true)}
            >
              Manual Entry
            </Button>
          </div>
        ) : (
          /* MANUAL ENTRY MODE (inline edit — no layout shift) */
          <FormField
            control={form.control}
            name="jobNumber"
            render={({ field }) => (
              <FormItem>
                <div className="flex items-center gap-3">
                  <FormControl>
                    <Input
                      {...field}
                      value={field.value || ""}
                      className="font-mono w-48"
                      maxLength={50}
                      onChange={(e) => {
                        field.onChange(e);
                        if (duplicateProject || form.formState.errors.jobNumber) {
                          form.clearErrors("jobNumber");
                          setDuplicateProject(null);
                        }
                      }}
                      onBlur={(e) => {
                        field.onBlur();
                        handleJobNumberBlur(e.target.value);
                      }}
                    />
                  </FormControl>
                  {jobNumberSettings.enabled && (
                    <Button
                      type="button"
                      variant="outline"
                      size="sm"
                      onClick={() => {
                        setManualJobNumber(false);
                        field.onChange(undefined);
                        form.clearErrors("jobNumber");
                        setDuplicateProject(null);
                      }}
                    >
                      Use Auto-Generated
                    </Button>
                  )}
                </div>
                <FormMessage />
                {isCheckingDuplicate && (
                  <div className="flex items-center gap-1.5 text-xs text-muted-foreground mt-1">
                    <Loader2 className="h-3 w-3 animate-spin" />
                    Checking availability...
                  </div>
                )}
                {duplicateProject && !isCheckingDuplicate && (
                  <div className="flex items-center gap-1 text-xs text-destructive mt-1">
                    <span className="font-mono font-medium">{form.getValues("jobNumber")}</span>
                    <span>is already assigned to</span>
                    <a
                      href={`/projects/${duplicateProject.slug}`}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="font-medium underline hover:no-underline inline-flex items-center gap-0.5"
                    >
                      {duplicateProject.name}
                      <ExternalLink className="h-3 w-3" />
                    </a>
                  </div>
                )}
              </FormItem>
            )}
          />
        )}
      </div>
    )}

    {/* PROJECT NAME */}
    <FormField
      control={form.control}
      name="name"
      render={({ field }) => (
        <FormItem>
          <FormLabel>Project Name</FormLabel>
          <FormControl>
            <Input placeholder="Downtown Office Tower" {...field} />
          </FormControl>
          <FormMessage />
        </FormItem>
      )}
    />

    {/* CUSTOMER */}
    <FormField
      control={form.control}
      name="customerId"
      render={({ field }) => (
        <FormItem>
          <FormLabel>Customer *</FormLabel>
          <div className="flex gap-2">
            <Select value={field.value} onValueChange={field.onChange}>
              <FormControl>
                <SelectTrigger className="flex-1">
                  <SelectValue placeholder="Select customer" />
                </SelectTrigger>
              </FormControl>
              <SelectContent>
                {customers.map((customer) => (
                  <SelectItem key={customer.id} value={customer.id}>
                    {getCustomerDisplayName(customer)}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
            <Button type="button" variant="outline" onClick={() => setAddCustomerOpen(true)}>
              <Plus className="h-4 w-4" />
            </Button>
          </div>
          <FormMessage />
        </FormItem>
      )}
    />

    {/* DESCRIPTION */}
    <FormField
      control={form.control}
      name="description"
      render={({ field }) => (
        <FormItem>
          <FormLabel>Description</FormLabel>
          <FormControl>
            <Textarea placeholder="Brief project description..." className="min-h-[80px]" {...field} />
          </FormControl>
          <FormMessage />
        </FormItem>
      )}
    />

    {/* LOCATION — AddressInput, no heading above it */}
    <AddressInput form={form} prefix="location" />

    {/* BUILDING DIMENSIONS */}
    <div className="space-y-3">
      <p className="text-sm font-medium">Building Dimensions</p>
      <div className="grid grid-cols-3 gap-4">
        <FormField
          control={form.control}
          name="buildingWidth"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Width</FormLabel>
              <div className="flex items-center gap-1.5">
                <FormControl>
                  <Input
                    type="number"
                    min="0"
                    step="any"
                    placeholder="0"
                    {...field}
                    value={field.value ?? ""}
                    onChange={(e) => {
                      const val = e.target.value === "" ? undefined : Number(e.target.value);
                      field.onChange(val);
                      setBuildingWidth(val ?? "");
                    }}
                  />
                </FormControl>
                <span className="text-sm text-muted-foreground whitespace-nowrap">ft</span>
              </div>
              <FormMessage />
            </FormItem>
          )}
        />
        <FormField
          control={form.control}
          name="buildingLength"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Length</FormLabel>
              <div className="flex items-center gap-1.5">
                <FormControl>
                  <Input
                    type="number"
                    min="0"
                    step="any"
                    placeholder="0"
                    {...field}
                    value={field.value ?? ""}
                    onChange={(e) => {
                      const val = e.target.value === "" ? undefined : Number(e.target.value);
                      field.onChange(val);
                      setBuildingLength(val ?? "");
                    }}
                  />
                </FormControl>
                <span className="text-sm text-muted-foreground whitespace-nowrap">ft</span>
              </div>
              <FormMessage />
            </FormItem>
          )}
        />
        <FormField
          control={form.control}
          name="buildingHeight"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Height</FormLabel>
              <div className="flex items-center gap-1.5">
                <FormControl>
                  <Input
                    type="number"
                    min="0"
                    step="any"
                    placeholder="0"
                    {...field}
                    value={field.value ?? ""}
                    onChange={(e) => {
                      const val = e.target.value === "" ? undefined : Number(e.target.value);
                      field.onChange(val);
                    }}
                  />
                </FormControl>
                <span className="text-sm text-muted-foreground whitespace-nowrap">ft</span>
              </div>
              <FormMessage />
            </FormItem>
          )}
        />
      </div>
      {/* Live sq ft preview */}
      {buildingWidth !== "" && buildingLength !== "" && (
        <p className="text-sm text-muted-foreground">
          Sq ft:{" "}
          <span className="font-medium text-foreground">
            {(Number(buildingWidth) * Number(buildingLength)).toLocaleString()}
          </span>
        </p>
      )}
    </div>

    {/* CUSTOMER BUDGET */}
    <FormField
      control={form.control}
      name="customerBudget"
      render={({ field }) => (
        <FormItem>
          <FormLabel>Customer Budget</FormLabel>
          <FormControl>
            <CurrencyInput
              value={field.value ?? 0}
              onChange={(val) => field.onChange(val === 0 ? undefined : val)}
            />
          </FormControl>
          <FormMessage />
        </FormItem>
      )}
    />

    {/* SUBMIT */}
    <div className="flex justify-end gap-3">
      <Button type="button" variant="outline" onClick={() => router.back()}>
        Cancel
      </Button>
      <Button
        type="submit"
        disabled={createMutation.isPending || isCheckingDuplicate || !!duplicateProject}
      >
        {createMutation.isPending ? "Creating..." : "Create Project"}
      </Button>
    </div>

  </form>
  {/* Add Customer Dialog — keep unchanged */}
  ...
</Form>
```

- The Project Number widget is rendered OUTSIDE the Card component (no Card wrapper). It is a standalone `<div className="space-y-1.5">` directly in the form.
- No Card components wrap any fields. The form is a flat list of field groups with `className="space-y-6"` on the `<form>` tag.
- The Add Customer Dialog at the bottom of the component remains completely unchanged.
- Remove the `Project Type`, `Start Date`, and `End Date` fields from the form. They are not in the new field order per CONTEXT.md. (The schema still accepts them; they just won't be on the create form.)
- `costCodes` stays in defaultValues as `[]` but is not rendered as a UI section. The server will store `[]`.
- The `INFO` icon, `FormDescription`, and tooltip imports can be removed if they are no longer used anywhere in the component after removing the old Job Number card section.
