    **Blur validation — identical pattern to project-form.tsx but with `excludeProjectId`:**
    ```typescript
    const [isCheckingDuplicate, setIsCheckingDuplicate] = useState(false);
    const [duplicateProject, setDuplicateProject] = useState<{ name: string; slug: string } | null>(null);

    const handleJobNumberBlur = async (value: string) => {
      setDuplicateProject(null);
      if (!value) { form.clearErrors("jobNumber"); return; }

      // Format check (only when auto-gen is enabled)
      if (jobNumberSettings?.enabled) {
        if (!validateJobNumberFormat(value, jobNumberSettings.template)) {
          form.setError("jobNumber", { type: "manual", message: "Job number doesn't match the required format" });
          return;
        }
      }

      // Duplicate check (exclude THIS project's own ID)
      form.clearErrors("jobNumber");
      setIsCheckingDuplicate(true);
      try {
        const result = await utils.company.checkJobNumberAvailability.fetch({
          jobNumber: value,
          excludeProjectId: project?.id,  // KEY: prevents self-match
        });
        if (!result.available && result.existingProjectName && result.existingProjectSlug) {
          setDuplicateProject({ name: result.existingProjectName, slug: result.existingProjectSlug });
          form.setError("jobNumber", { type: "manual", message: " " });
        }
      } catch {
        // Don't block on network error
      } finally {
        setIsCheckingDuplicate(false);
      }
    };
    ```

    **Save mutation:**
    ```typescript
    const updateMutation = trpc.project.update.useMutation({

        utils.project.getBySlug.invalidate({ slug: projectSlug });

        toast.success("Project updated");
        router.push(`/projects/${projectSlug}`);
      },
      onError: (error) => {
        toast.error(error.message || "Failed to update project");

    });
    ```

    ```typescript
    const onSubmit = async (values: z.infer<typeof updateProjectSchema>) => {
    ```
