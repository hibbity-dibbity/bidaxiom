```typescript
import { getIronSession, type SessionOptions } from "iron-session";
import { cookies } from "next/headers";

export interface SessionData {
  userId?: string;
  accessToken?: string;
  refreshToken?: string;
  orgId?: string;
  orgRole?: string;
  orgName?: string;
  hasMfa?: boolean;
  user?: {
    email: string;
    firstName?: string | null;
    lastName?: string | null;
  };
}

export const sessionOptions: SessionOptions = {
  password: process.env.SESSION_SECRET!,
  cookieName: "construction_session",
  cookieOptions: {
    secure: process.env.NODE_ENV === "production",
    httpOnly: true,
    sameSite: "lax" as const, // "lax" allows OAuth redirects, "strict" blocks them
    maxAge: 7 * 24 * 60 * 60, // 7 days (matches refresh token TTL)
  },
};

export async function getSession() {
  return await getIronSession<SessionData>(await cookies(), sessionOptions);
}
```

```typescript
"use server";
// Server Actions for auth operations

import { getSession } from "./session";
import { redirect } from "next/navigation";

const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:3001";

/**
 * Get the current access token from session.
 * If expired, attempt to refresh using the refresh token.
 */
export async function getAccessToken(): Promise<string | null> {
  const session = await getSession();
  if (!session.accessToken) return null;

  // Try to decode the JWT to check expiry (without verification -- that's the API's job)
  try {
    const payload = JSON.parse(
      Buffer.from(session.accessToken.split(".")[1], "base64").toString()
    );
    const expiresAt = payload.exp * 1000;
    const now = Date.now();

    // If token expires in less than 2 minutes, refresh proactively
    if (expiresAt - now < 2 * 60 * 1000) {
      return await refreshAccessToken();
    }

    return session.accessToken;
  } catch {
    return session.accessToken; // If decode fails, let the API handle it
  }
}

/**
 * Refresh the access token using the refresh token.
 */
export async function refreshAccessToken(): Promise<string | null> {
  const session = await getSession();
  if (!session.refreshToken) return null;

  try {
    const response = await fetch(`${API_URL}/trpc/auth.refresh`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        json: { refreshToken: session.refreshToken },
      }),
    });

    if (!response.ok) {
      // Refresh failed -- clear session and force re-login
      session.destroy();
      return null;
    }

    const data = await response.json();
    const result = data.result?.data?.json;

    if (result?.accessToken && result?.refreshToken) {
      session.accessToken = result.accessToken;
      session.refreshToken = result.refreshToken;
      await session.save();
      return result.accessToken;
    }

    return null;
  } catch {
    return null;
  }
}

/**
 * Log in with credentials. Stores tokens in session.
 */
export async function login(data: {
  accessToken: string;
  refreshToken: string;
  user: { id: string; email: string; firstName?: string | null; lastName?: string | null; mfaEnabled: boolean };
  org?: { id: string; name: string; role: string } | null;
}) {
  const session = await getSession();
  session.userId = data.user.id;
  session.accessToken = data.accessToken;
  session.refreshToken = data.refreshToken;
  session.hasMfa = data.user.mfaEnabled;
  session.user = {
    email: data.user.email,
    firstName: data.user.firstName,
    lastName: data.user.lastName,
  };
  if (data.org) {
    session.orgId = data.org.id;
    session.orgRole = data.org.role;
    session.orgName = data.org.name;
  }
  await session.save();
}

/**
 * Log out. Destroys session.
 */
export async function logout() {
  const session = await getSession();
  session.destroy();
  redirect("/sign-in");
}

/**
 * Update the active organization in session.
 */
export async function setActiveOrg(org: { id: string; name: string; role: string }, accessToken: string) {
  const session = await getSession();
  session.orgId = org.id;
  session.orgRole = org.role;
  session.orgName = org.name;
  session.accessToken = accessToken;
  await session.save();
}

/**
 * Get session data for client components (read-only snapshot).
 */
export async function getSessionData(): Promise<SessionData> {
  const session = await getSession();
  return {
    userId: session.userId,
    orgId: session.orgId,
    orgRole: session.orgRole,
    orgName: session.orgName,
    hasMfa: session.hasMfa,
    user: session.user,
    // Do NOT return tokens to client
  };
}
```

```typescript
"use client";

import { createContext, useContext, useState, useEffect, useCallback, useMemo, type ReactNode } from "react";
import { defineAbilitiesFor } from "@/lib/abilities";
import type { SessionData } from "./session";

interface AuthContextType {
  userId: string | null;
  orgId: string | null;
  orgRole: string | null;
  orgName: string | null;
  hasMfa: boolean;
  user: { email: string; firstName?: string | null; lastName?: string | null } | null;
  isLoaded: boolean;
  // These will be called by components; they trigger server actions via fetch
  switchOrg: (orgId: string) => Promise<void>;
}

const AuthContext = createContext<AuthContextType>({
  userId: null,
  orgId: null,
  orgRole: null,
  orgName: null,
  hasMfa: false,
  user: null,
  isLoaded: false,
  switchOrg: async () => {},
});

export function AuthProvider({
  children,
  initialSession,
}: {
  children: ReactNode;
  initialSession: SessionData;
}) {
  const [session, setSession] = useState<SessionData>(initialSession);
  const [isLoaded, setIsLoaded] = useState(true); // Already loaded from server

  const switchOrg = useCallback(async (orgId: string) => {
    // Call the API to switch org and get new token
    // Then update session via server action
    try {
      const response = await fetch("/api/auth/switch-org", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ organizationId: orgId }),
      });
      if (response.ok) {
        const data = await response.json();
        setSession(prev => ({
          ...prev,
          orgId: data.org.id,
          orgRole: data.org.role,
          orgName: data.org.name,
        }));
        // Reload to refresh all data with new org context
        window.location.reload();
      }
    } catch (err) {
      console.error("Failed to switch org:", err);
    }
  }, []);

  const value = useMemo(() => ({
    userId: session.userId ?? null,
    orgId: session.orgId ?? null,
    orgRole: session.orgRole ?? null,
    orgName: session.orgName ?? null,
    hasMfa: session.hasMfa ?? false,
    user: session.user ?? null,
    isLoaded,
    switchOrg,
  }), [session, isLoaded, switchOrg]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  return useContext(AuthContext);
}
```

Replace Clerk middleware entirely:

```typescript
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";
import { getIronSession } from "iron-session";
import { type SessionData, sessionOptions } from "./lib/session";

const publicRoutes = ["/sign-in", "/sign-up", "/verify-email", "/reset-password", "/api/webhooks"];
const mfaSetupRoute = "/mfa-setup";

function isPublicRoute(pathname: string): boolean {
  return publicRoutes.some((route) => pathname.startsWith(route));
}

const ENFORCE_MFA = process.env.NEXT_PUBLIC_ENFORCE_MFA === "true";

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Allow public routes
  if (isPublicRoute(pathname)) {
    return NextResponse.next();
  }

  // Allow static files and Next.js internals
  if (pathname.startsWith("/_next") || pathname.includes(".")) {
    return NextResponse.next();
  }

  // Get session from cookie
  const response = NextResponse.next();
  const session = await getIronSession<SessionData>(request, response, sessionOptions);

  // No authenticated user -> redirect to sign-in
  if (!session.userId) {
    const signInUrl = new URL("/sign-in", request.url);
    signInUrl.searchParams.set("redirect_url", pathname);
    return NextResponse.redirect(signInUrl);
  }

  // MFA enforcement
  if (ENFORCE_MFA && !session.hasMfa && !pathname.startsWith(mfaSetupRoute)) {
    return NextResponse.redirect(new URL(mfaSetupRoute, request.url));
  }

  return response;
}

export const config = {
  matcher: ["/((?!.*\\..*|_next).*)", "/", "/(api|trpc)(.*)"],
};
```

```typescript
"use client";

import { useState, useMemo } from "react";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { httpBatchLink } from "@trpc/react-query";
import { trpc } from "@/lib/trpc";
import { Toaster } from "@/components/ui/sonner";
import { AbilityContext, defineAbilitiesFor } from "@/lib/abilities";
import { AuthProvider, useAuth } from "@/lib/auth-context";
import type { SessionData } from "@/lib/session";

function TrpcAndAbilityProvider({ children }: { children: React.ReactNode }) {
  const { orgRole, userId } = useAuth();

  // Build CASL abilities from org role
  const abilities = useMemo(() => {
    const role = orgRole ?? "org:member";
    return defineAbilitiesFor(role, userId ?? "");
  }, [orgRole, userId]);

  const [queryClient] = useState(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: {
            staleTime: 5 * 60 * 1000,
            retry: 1,
            refetchOnWindowFocus: false,
          },
        },
      })
  );

  const [trpcClient] = useState(() =>
    trpc.createClient({
      links: [
        httpBatchLink({
          url: `${process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:3001"}/trpc`,
          async headers() {
            // Fetch access token from session via API route
            try {
              const res = await fetch("/api/auth/token");
              if (res.ok) {
                const { accessToken } = await res.json();
                return accessToken ? { Authorization: `Bearer ${accessToken}` } : {};
              }
            } catch {}
            return {};
          },
        }),
      ],
    })
  );

  return (
    <trpc.Provider client={trpcClient} queryClient={queryClient}>
      <QueryClientProvider client={queryClient}>
        <AbilityContext.Provider value={abilities}>
          {children}
          <Toaster />
        </AbilityContext.Provider>
      </QueryClientProvider>
    </trpc.Provider>
  );
}

export function Providers({
  children,
  initialSession,
}: {
  children: React.ReactNode;
  initialSession: SessionData;
}) {
  return (
    <AuthProvider initialSession={initialSession}>
      <TrpcAndAbilityProvider>{children}</TrpcAndAbilityProvider>
    </AuthProvider>
  );
}
```

```typescript
import { NextResponse } from "next/server";
import { setActiveOrg } from "@/lib/auth";

export async function POST(request: Request) {
  const { organizationId } = await request.json();
  // Call API to switch org
  const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:3001";
  const { getAccessToken } = await import("@/lib/auth");
  const token = await getAccessToken();

  const response = await fetch(`${API_URL}/trpc/auth.switchOrganization`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${token}`,
    },
    body: JSON.stringify({ json: { organizationId } }),
  });

  if (!response.ok) {
    return NextResponse.json({ error: "Failed to switch org" }, { status: 400 });
  }

  const data = await response.json();
  const result = data.result?.data?.json;

  if (result) {
    await setActiveOrg(result.org, result.accessToken);
    return NextResponse.json({ org: result.org });
  }

  return NextResponse.json({ error: "Invalid response" }, { status: 400 });
}
```
