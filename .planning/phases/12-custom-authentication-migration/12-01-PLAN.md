```typescript
// --- Custom Auth Tables (Phase 12) ---

/**
 * Users table - replaces Clerk user management.
 * Central user identity. Each user can belong to multiple organizations.
 */
export const users = pgTable("users", {
  id: uuid("id").primaryKey().defaultRandom(),
  email: text("email").notNull().unique(),
  emailVerified: boolean("email_verified").notNull().default(false),
  passwordHash: text("password_hash").notNull(),
  hashType: text("hash_type").notNull().default("argon2id"), // argon2id or bcrypt (for migration)
  firstName: text("first_name"),
  lastName: text("last_name"),
  avatarUrl: text("avatar_url"),
  mfaEnabled: boolean("mfa_enabled").notNull().default(false),
  mfaSecret: text("mfa_secret"), // Encrypted TOTP secret
  mfaBackupCodes: jsonb("mfa_backup_codes").$type<string[]>(), // Hashed backup codes
  clerkUserId: text("clerk_user_id"), // For migration tracking, nullable
  lastLoginAt: timestamp("last_login_at", { withTimezone: true }),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).defaultNow().notNull(),
}, (table) => ({
  emailIdx: index("idx_users_email").on(table.email),
  clerkUserIdx: index("idx_users_clerk_user_id").on(table.clerkUserId),
}));

/**
 * Organizations - replaces Clerk Organizations.
 * Each organization maps to a tenant schema.
 */
export const organizations = pgTable("organizations", {
  id: uuid("id").primaryKey().defaultRandom(),
  name: text("name").notNull(),
  slug: text("slug").notNull().unique(), // URL-safe identifier
  tenantId: uuid("tenant_id").references(() => tenants.id, { onDelete: "cascade" }),
  createdBy: uuid("created_by").references(() => users.id),
  clerkOrgId: text("clerk_org_id"), // For migration tracking, nullable
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).defaultNow().notNull(),
}, (table) => ({
  slugIdx: index("idx_organizations_slug").on(table.slug),
  clerkOrgIdx: index("idx_organizations_clerk_org_id").on(table.clerkOrgId),
}));

/**
 * Organization memberships - replaces Clerk organization memberships.
 * Maps users to organizations with roles.
 */
export const organizationMemberships = pgTable("organization_memberships", {
  id: uuid("id").primaryKey().defaultRandom(),
  userId: uuid("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  organizationId: uuid("organization_id").notNull().references(() => organizations.id, { onDelete: "cascade" }),
  role: text("role").notNull().default("org:member"), // org:admin, org:member, org:project_manager, etc.
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).defaultNow().notNull(),
}, (table) => ({
  userOrgUnique: unique().on(table.userId, table.organizationId),
  userIdx: index("idx_org_memberships_user").on(table.userId),
  orgIdx: index("idx_org_memberships_org").on(table.organizationId),
}));

/**
 * Organization invitations - replaces Clerk invitation flow.
 * Admin-initiated invitations with 7-day TTL.
 */
export const organizationInvitations = pgTable("organization_invitations", {
  id: uuid("id").primaryKey().defaultRandom(),
  organizationId: uuid("organization_id").notNull().references(() => organizations.id, { onDelete: "cascade" }),
  email: text("email").notNull(),
  role: text("role").notNull().default("org:member"),
  tokenHash: text("token_hash").notNull(), // Hashed invitation token
  status: text("status").notNull().default("pending"), // pending, accepted, expired, revoked
  invitedBy: uuid("invited_by").notNull().references(() => users.id),
  expiresAt: timestamp("expires_at", { withTimezone: true }).notNull(),
  acceptedAt: timestamp("accepted_at", { withTimezone: true }),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
}, (table) => ({
  emailIdx: index("idx_org_invitations_email").on(table.email),
  orgIdx: index("idx_org_invitations_org").on(table.organizationId),
}));

/**
 * Refresh tokens - for JWT refresh token rotation with reuse detection.
 * Stored in Redis for fast lookup; DB is fallback/audit.
 */
export const refreshTokens = pgTable("refresh_tokens", {
  id: uuid("id").primaryKey().defaultRandom(),
  userId: uuid("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  tokenHash: text("token_hash").notNull().unique(),
  familyId: text("family_id").notNull(), // Token family for rotation tracking
  expiresAt: timestamp("expires_at", { withTimezone: true }).notNull(),
  revokedAt: timestamp("revoked_at", { withTimezone: true }),
  replacedByHash: text("replaced_by_hash"), // Points to the token that replaced this one
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
}, (table) => ({
  userIdx: index("idx_refresh_tokens_user").on(table.userId),
  familyIdx: index("idx_refresh_tokens_family").on(table.familyId),
}));

/**
 * Email verification tokens.
 * Token is hashed before storage; plaintext sent in email.
 */
export const emailVerificationTokens = pgTable("email_verification_tokens", {
  id: uuid("id").primaryKey().defaultRandom(),
  userId: uuid("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  tokenHash: text("token_hash").notNull(),
  expiresAt: timestamp("expires_at", { withTimezone: true }).notNull(),
  usedAt: timestamp("used_at", { withTimezone: true }),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
}, (table) => ({
  userIdx: index("idx_email_verification_user").on(table.userId),
}));

/**
 * Password reset tokens.
 * Token is hashed before storage; plaintext sent in email.
 */
export const passwordResetTokens = pgTable("password_reset_tokens", {
  id: uuid("id").primaryKey().defaultRandom(),
  userId: uuid("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  tokenHash: text("token_hash").notNull(),
  expiresAt: timestamp("expires_at", { withTimezone: true }).notNull(),
  usedAt: timestamp("used_at", { withTimezone: true }),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
}, (table) => ({
  userIdx: index("idx_password_reset_user").on(table.userId),
}));
```

**Create `apps/api/src/auth/jwt.ts`:**
- Use `jose` library (NOT jsonwebtoken -- ESM issues, algorithm confusion CVEs)
- `signAccessToken(payload: { userId: string; orgId: string; orgRole: string; hasMfa: boolean })`: Returns JWT with HS256, 15-min expiry. Set `sub` to userId. Include orgId, orgRole, hasMfa in payload.
- `signRefreshToken(payload: { userId: string; tokenFamily: string })`: Returns JWT with HS256 using separate JWT_REFRESH_SECRET, 7-day expiry.
- `verifyAccessToken(token: string)`: Returns decoded payload or throws.
- `verifyRefreshToken(token: string)`: Returns decoded payload or throws.
- Secrets from `process.env.JWT_SECRET` and `process.env.JWT_REFRESH_SECRET`, both encoded via `new TextEncoder().encode()`.

**Create `apps/api/src/auth/password.ts`:**
- Use `@node-rs/argon2` (NOT bcrypt for new passwords -- OWASP recommends Argon2id)
- OWASP minimum config: `memoryCost: 19456` (19 MiB), `timeCost: 2`, `parallelism: 1`
- `hashPassword(password: string)`: Returns Argon2id hash string
- `verifyPassword(storedHash: string, password: string, hashType: 'argon2id' | 'bcrypt')`: For argon2id, use @node-rs/argon2 verify. For bcrypt (migrated Clerk users), use a bcrypt comparison import from @node-rs/argon2 if available, otherwise add bcrypt as dep. Return `{ valid: boolean; needsRehash: boolean }` where needsRehash=true when hashType is bcrypt and password is valid (caller should rehash with Argon2id).
- NOTE: For bcrypt verification of migrated Clerk hashes, check if @node-rs/argon2 handles bcrypt. If not, import a lightweight bcrypt verify. Since Clerk uses bcrypt, we need bcrypt verification capability. Use the `bcrypt` pattern from the hash string itself ($2a$, $2b$ prefix) to detect and verify accordingly. The `@node-rs/argon2` library only does Argon2. Add `bcryptjs` as a dependency for bcrypt verification only (lightweight, pure JS, no native compilation). Run: `cd apps/api && npm install bcryptjs && npm install -D @types/bcryptjs`

**Create `apps/api/src/auth/mfa.ts`:**
- Use `otpauth` (NOT speakeasy -- unmaintained 7+ years)
- `generateMfaSecret(userEmail: string)`: Creates TOTP with issuer "ConstructionMgmt", SHA1, 6 digits, 30s period. Returns `{ secret: string (base32), qrCodeDataUrl: string, backupCodes: string[], hashedBackupCodes: string[] }`. Generate 10 backup codes with `nanoid(10)`, hash each with Argon2id.
- `verifyMfaToken(secret: string, token: string)`: Verify TOTP with `window: 1` (accepts +/-30s drift). Returns boolean.
- `verifyBackupCode(hashedCodes: string[], code: string)`: Try verifying code against each hashed backup code. Return `{ valid: boolean; remainingCodes: string[] }` (remove used code from array).
- Use `qrcode` for QR code data URL generation via `QRCode.toDataURL(totp.toString())`.

**Create `apps/api/src/auth/tokens.ts`:**
- Use `nanoid` for cryptographically random token generation
- Use `createHash` from node:crypto for SHA-256 hashing (fast, non-reversible, suitable for token lookup -- NOT password storage)
- `generateToken(length?: number)`: Returns `{ token: string, hash: string }`. Default length 32. Token is the plaintext (sent to user), hash is SHA-256 hex (stored in DB).
- `hashToken(token: string)`: Returns SHA-256 hex hash. Used for verification lookup.
- Note: We use SHA-256 for tokens (not Argon2id) because tokens are high-entropy random strings, not user-chosen passwords. SHA-256 is sufficient and much faster for lookup operations.

**Create `apps/api/src/auth/rate-limit.ts`:**
- Use `rate-limiter-flexible` with Redis backend
- Import existing Redis connection pattern from `apps/api/src/integrations/qbo/queue.ts` (or wherever ioredis is configured). If no shared Redis client exists, create one using `process.env.REDIS_URL` with lazy initialization (same pattern as the DB pool proxy).
- Create 3 limiters:
  1. `loginLimiter`: 5 attempts per 15 minutes per username+IP key. Block for 15 min.
  2. `registrationLimiter`: 3 attempts per hour per IP. Block for 1 hour.
  3. `refreshLimiter`: 10 attempts per minute per userId. Block for 1 min.
- Export `checkLoginRateLimit(email: string, ip: string)`, `checkRegistrationRateLimit(ip: string)`, `checkRefreshRateLimit(userId: string)` functions that throw TRPCError with code TOO_MANY_REQUESTS on limit exceeded.
- Handle Redis not available gracefully (log warning, allow request -- non-fatal, same pattern as Phase 8 QBO worker).
