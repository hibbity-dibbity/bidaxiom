**1. Create apps/web/app/(dashboard)/projects/components/delete-project-button.tsx**

This is a compound component managing three states: idle, blocking dialog open, confirmation dialog open.

```tsx
"use client";

import React, { useState } from "react";
import { trpc } from "@/lib/trpc";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import {
  AlertDialog,
  AlertDialogContent,
  AlertDialogHeader,
  AlertDialogTitle,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogCancel,
} from "@/components/ui/alert-dialog";
import { Textarea } from "@/components/ui/textarea";
import { Trash2 } from "lucide-react";

interface DeleteProjectButtonProps {
  projectId: string;
  projectName: string;
  onDeleted?: () => void;
  variant?: "table-row" | "menu-item";  // controls rendering style
}
```

**State machine:**
- `"idle"` — nothing open
- `"checking"` — checkDeletable mutation running (show loading state on button)
- `"blocked"` — BlockedDialog open (show reasons list)
- `"confirming"` — ConfirmDialog open (show reason textarea)

**checkDeletable flow:** On button click → call `trpc.project.checkDeletable.useMutation()`. On success: if `!canDelete` → set state to `"blocked"` with reasons. If `canDelete` → set state to `"confirming"`.

**Blocked dialog content:**
- Title: "Cannot Delete Project"
- Body: "This project cannot be deleted for the following reasons:"
- Unordered list of `reasons` strings
- Single "Close" button (AlertDialogCancel)

**Confirmation dialog content:**
- Title: "Delete Project"
- Body: "This action cannot be easily undone. The project will be permanently deleted after 30 days. You can recover it within 30 days via Company Settings → Deleted Projects."
- Label + Textarea for "Reason for deletion" (required, `value={reason}` controlled)
- Footer: Cancel button + Delete button (destructive variant, `disabled={reason.trim().length === 0}`, calls `softDelete.mutate({ projectId, reason })`)
- On softDelete success: toast.success("Project deleted"), call `onDeleted?.()`, reset state to "idle"
- On softDelete error: toast.error(error.message)

**Render for `variant="table-row"`:** Standalone `<Button variant="destructive" size="sm">` with Trash2 icon only (no text), `className="opacity-0 group-hover:opacity-100 transition-opacity"`. The parent `TableRow` must have `className="group"`.

**Render for `variant="menu-item"`:** A `<button>` styled as a dropdown item with Trash2 icon + "Delete Project" text, using destructive text color (e.g., `className="text-destructive flex items-center gap-2 w-full px-2 py-1.5 text-sm rounded hover:bg-accent"`).

**2. Create apps/web/app/(dashboard)/settings/deleted-projects/page.tsx:**

```tsx
"use client";

import { trpc } from "@/lib/trpc";
import { Button } from "@/components/ui/button";
import { toast } from "sonner";

export default function DeletedProjectsPage() {
  const utils = trpc.useUtils();
  const { data, isLoading } = trpc.project.listDeleted.useQuery();

  const restoreMutation = trpc.project.restore.useMutation({
    onSuccess: () => {
      utils.project.listDeleted.invalidate();
      toast.success("Project restored. It will show a restoration banner.");
    },
    onError: (error) => toast.error(error.message),
  });

  if (isLoading) return <div className="p-6 text-muted-foreground">Loading...</div>;

  return (
    <div className="space-y-6">
      <div>
        <h2 className="text-xl font-semibold">Deleted Projects</h2>
        <p className="text-sm text-muted-foreground mt-1">
          Deleted projects are permanently removed after 30 days. Restore a project to recover all data.
        </p>
      </div>

      {(!data || data.length === 0) ? (
        <div className="rounded border p-8 text-center text-muted-foreground text-sm">
          No deleted projects
        </div>
      ) : (
        <div className="rounded border divide-y">
          {data.map((project) => {
            const deletedAtDate = project.deletedAt ? new Date(project.deletedAt) : null;
            const formattedDate = deletedAtDate
              ? deletedAtDate.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" })
              : "—";
            return (
              <div key={project.id} className="flex items-center justify-between px-4 py-3">
                <div className="space-y-0.5">
                  <p className="font-medium text-sm">{project.name}</p>
                  <p className="text-xs text-muted-foreground">
                    Deleted {formattedDate}
                    {project.deletedByName ? ` by ${project.deletedByName}` : ""}
                    {" · "}
                    {project.daysRemaining} day{project.daysRemaining !== 1 ? "s" : ""} remaining
                  </p>
                  {project.deletionReason && (
                    <p className="text-xs text-muted-foreground italic">"{project.deletionReason}"</p>
                  )}
                </div>
                <Button
                  variant="outline"
                  size="sm"
                  onClick={() => restoreMutation.mutate({ projectId: project.id })}
                  disabled={restoreMutation.isPending}
                >
                  Restore
                </Button>
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}
```
